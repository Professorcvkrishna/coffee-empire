    function advanceTurn() {
      const sim = window.APP_STATE.sim;
      const activeLocs = Object.values(sim.locations).filter(l => l.isOpen);
      if (activeLocs.length === 0) {
        showToast("Open at least one coffee shop location!", 'error');
        return;
      }

      let qtrRev = 0, qtrProf = 0, qtrCups = 0;
      const currentEnvMod = sim.marketEvents[Math.min(3, sim.year)].modifiers;

      Object.keys(sim.locations).forEach(key => {
        const loc = sim.locations[key];
        if (!loc.isOpen) return;

        const d = loc.decisions;
        const envMod = currentEnvMod[key] || 1.0;

        // Staff Capacity & Morale
        const baseWageRatio = d.wage / 16.0;
        loc.stats.morale = Math.min(100, Math.max(30, Math.floor(70 * baseWageRatio)));
        const nightMult = d.schedule === '24h' ? 1.6 : 1.0;
        const totalCap = d.staff * 2200 * nightMult * (loc.stats.morale / 100);

        // Demand
        const priceRatio = d.price / 4.0;
        const priceDemandFactor = Math.max(0.2, 1.8 - (priceRatio * loc.priceSensitivity));
        const promoQuarter = d.promo / 4.0;
        loc.stats.awareness = Math.min(100, loc.stats.awareness + (promoQuarter / 500));

        let demandCups = loc.baseTraffic * envMod * priceDemandFactor * (loc.stats.awareness / 50) * nightMult;
        const cupsSold = Math.floor(Math.min(demandCups, totalCap));

        const serviceRate = demandCups > 0 ? Math.min(1.0, totalCap / demandCups) : 1.0;
        loc.stats.satisfaction = Math.min(100, Math.max(20, Math.floor((serviceRate * 60) + (loc.stats.morale * 0.4))));

        // Financials
        const coffeeCost = 0.80 * (sim.year === 2 ? 1.3 : 1.0);
        const grossRev = cupsSold * d.price;
        const cogs = cupsSold * coffeeCost;
        const laborCost = d.staff * (d.wage * 160 * 3) * (d.schedule === '24h' ? 1.5 : 1.0);
        const rent = loc.rentPerYear / 4.0;
        const marketing = d.promo / 4.0;

        const netProf = grossRev - (cogs + laborCost + rent + marketing);

        loc.stats.lastQuarterRevenue = grossRev;
        loc.stats.lastQuarterProfit = netProf;
        loc.stats.lastQuarterCups = cupsSold;

        qtrRev += grossRev;
        qtrProf += netProf;
        qtrCups += cupsSold;
      });

      sim.totalCapital += qtrProf;
      sim.cumulativeRevenue += qtrRev;

      const labelStr = `Y${sim.year}Q${sim.quarter}`;
      sim.history.labels.push(labelStr);
      sim.history.revenue.push(qtrRev);
      sim.history.profit.push(qtrProf);
      sim.history.cups.push(qtrCups);

      sim.quarter++;
      if (sim.quarter > 4) {
        sim.quarter = 1;
        sim.year++;
      }

      updateCharts();
      updateUI();
      syncCurrentStudentDataToFirebase(false);

      if (sim.year > 3) {
        showToast("Simulation Complete! Review your report card.", 'success');
        switchStudentSubTab('report');
      } else {
        showToast(`Quarter ${labelStr} completed! Profit: $${Math.floor(qtrProf).toLocaleString()}`, 'info');
      }
    }

    // --- DATA PERSISTENCE & INSTRUCTOR CLOUD SYNCHRONIZATION ---
    async function syncCurrentStudentDataToFirebase(manualToast = false) {
      if (window.APP_STATE.currentUserRole !== 'student') return;

      const info = window.APP_STATE.studentInfo;
      const sim = window.APP_STATE.sim;
      if (!info.id) return;

      const payload = {
        studentName: info.name,
        studentId: info.id,
        classCode: info.classCode,
        timestamp: new Date().toISOString(),
        finalCapital: Math.floor(sim.totalCapital),
        cumulativeRevenue: Math.floor(sim.cumulativeRevenue),
        totalProfit: Math.floor(sim.history.profit.reduce((a, b) => a + b, 0)),
        turnsCompleted: sim.history.labels.length,
        locations: sim.locations,
        history: sim.history
      };

      // 1. Save to Local Persistence
      window.APP_STATE.allSubmissions[info.id] = payload;
      localStorage.setItem('coffee_empire_submissions', JSON.stringify(window.APP_STATE.allSubmissions));

      // 2. Save to Firebase Firestore if online
      if (window.firebaseActive && window.db) {
        try {
          const docRef = doc(window.db, 'artifacts', window.appId, 'public', 'data', 'submissions', info.id);
          await setDoc(docRef, payload);
          if (manualToast) showToast('Synced directly to Instructor Cloud Database!', 'success');
        } catch (e) {
          console.error("Firestore Sync Error:", e);
          if (manualToast) showToast('Saved to local storage fallback.', 'info');
        }
      } else if (manualToast) {
        showToast('Saved to local storage fallback.', 'info');
      }
    }

    function subscribeToInstructorData() {
      // 1. Load Local Persistence First
      loadLocalMockSubmissions();
      renderInstructorTable();

      // 2. Real-time Firebase Sync
      if (window.firebaseActive && window.db) {
        try {
          const colRef = collection(window.db, 'artifacts', window.appId, 'public', 'data', 'submissions');
          onSnapshot(colRef, (snapshot) => {
            snapshot.forEach(docSnap => {
              const data = docSnap.data();
              window.APP_STATE.allSubmissions[data.studentId] = data;
            });
            renderInstructorTable();
          }, (err) => console.error("Firestore Listen Error:", err));
        } catch (e) {
          console.error("Firebase Subscription Failed:", e);
        }
      }
    }

    function renderInstructorTable() {
      const submissions = Object.values(window.APP_STATE.allSubmissions);
      const search = (document.getElementById('input-search-student')?.value || '').toLowerCase();
      const sectionFilter = (document.getElementById('input-filter-section')?.value || '').toUpperCase();

      const filtered = submissions.filter(s => {
        const matchSearch = s.studentName.toLowerCase().includes(search) || s.studentId.toLowerCase().includes(search);
        const matchSection = !sectionFilter || sectionFilter === 'ALL' || s.classCode.toUpperCase() === sectionFilter;
        return matchSearch && matchSection;
      });

      // Update KPI Cards
      document.getElementById('inst-kpi-count').innerText = filtered.length;
      const totalProf = filtered.reduce((a, b) => a + (b.totalProfit || 0), 0);
      document.getElementById('inst-kpi-avg-profit').innerText = `$${filtered.length ? Math.floor(totalProf / filtered.length).toLocaleString() : 0}`;
      
      const topCap = filtered.reduce((max, s) => Math.max(max, s.finalCapital || 0), 0);
      document.getElementById('inst-kpi-top-cap').innerText = `$${topCap.toLocaleString()}`;

      // Table Render
      const tbody = document.getElementById('inst-tbody-students');
      if (filtered.length === 0) {
        tbody.innerHTML = `<tr><td colspan="8" class="p-6 text-center text-slate-500">No matching student submissions found.</td></tr>`;
        return;
      }

      tbody.innerHTML = filtered.map(s => {
        let badgeClass = s.finalCapital > 180000 ? 'text-emerald-400 bg-emerald-950/80 border-emerald-800' : 
                         s.finalCapital > 110000 ? 'text-amber-400 bg-amber-950/80 border-amber-800' : 
                         'text-rose-400 bg-rose-950/80 border-rose-800';

          vx: (Math.random() - 0.5) * 1.2,
          vy: (Math.random() - 0.5) * 0.8,
          color: Math.random() > 0.5 ? '#38bdf8' : '#a855f7'
        });
      }
    }

    function animateCanvas() {
      if (ctx && canvas && window.APP_STATE && window.APP_STATE.sim) {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        const sim = window.APP_STATE.sim;
        const selLoc = sim.locations[sim.selectedLocKey];

        // Room Background
        ctx.fillStyle = selLoc.decisions.schedule === '24h' ? '#020617' : '#1c130d';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        // Counter
        ctx.fillStyle = '#2b1e16';
        ctx.fillRect(40, 70, canvas.width - 80, 14);

        // Baristas
        if (selLoc.isOpen) {
          const staffCount = selLoc.decisions.staff;
          const spacing = (canvas.width - 100) / Math.max(1, staffCount);
          for (let i = 0; i < staffCount; i++) {
            ctx.beginPath();
            ctx.arc(60 + i * spacing, 58, 6, 0, Math.PI * 2);
            ctx.fillStyle = '#f59e0b';
            ctx.fill();
          }

          // Customers
          customers.forEach(c => {
            c.x += c.vx; c.y += c.vy;
            if (c.x < 40 || c.x > canvas.width - 40) c.vx *= -1;
            if (c.y < 85 || c.y > canvas.height - 15) c.vy *= -1;

            ctx.beginPath();
            ctx.arc(c.x, c.y, 4, 0, Math.PI * 2);
            ctx.fillStyle = c.color;
            ctx.fill();
          });
        }
      }
      requestAnimationFrame(animateCanvas);
    }
  </script>
